class Solution:
    def isHappy(self, n: int) -> bool:
        seen=set()
        while n!=1:
            if n in seen:
                return False
            seen.add(n)

            s=0
            temp=n
            while temp>0:
                d=temp%10
                s +=d*d
                temp //=10
            n=s
        return True
